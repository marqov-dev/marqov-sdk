cmake_minimum_required(VERSION 3.20)
project(qristal_python312_binding LANGUAGES CXX)
set(PYBIND11_FINDPYTHON ON)
find_package(Python 3.12 EXACT COMPONENTS Interpreter Development.Module REQUIRED)
find_package(pybind11 CONFIG REQUIRED)
find_package(qristal_core CONFIG REQUIRED)
file(GLOB bindings CONFIGURE_DEPENDS source/python/*.cpp)
pybind11_add_module(core NO_EXTRAS ${bindings})
target_compile_features(core PRIVATE cxx_std_20)
target_link_libraries(core PRIVATE qristal::core)
set_target_properties(core PROPERTIES BUILD_RPATH "/work/install-core/lib;/work/install-xacc/lib")
install(TARGETS core DESTINATION /opt/python312-bindings/qristal)
